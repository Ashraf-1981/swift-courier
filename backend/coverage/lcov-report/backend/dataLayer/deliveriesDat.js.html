const pool = require('../db');
 
// used by create delivery: data keys must match the column names
async function createDelivery(data) {
  const [result] = await pool.query('INSERT INTO deliveries SET ?', [data]);
  return result.insertId; // the new delivery's id
}
 
async function getDeliveryById(id) {
  const [rows] = await pool.query('SELECT * FROM deliveries WHERE id = ?', [id]);
  return rows[0]; // undefined if not found
}
 
// used by "My deliveries": one customer's list, with names instead of ids
async function getDeliveriesByCustomer(customerId) {
  const [rows] = await pool.query(
    `SELECT d.id, d.tracking_no, d.status, d.total_fee,
            d.scheduled_pickup_at, d.deadline_at, d.created_at,
            t.name  AS vehicle,
            pz.name AS pickup_zone,
            dz.name AS dropoff_zone,
            p.method AS payment_method,
            p.status AS payment_status
     FROM deliveries d
     JOIN delivery_types t ON t.id = d.delivery_type_id
     JOIN zones pz ON pz.id = d.pickup_zone_id
     JOIN zones dz ON dz.id = d.dropoff_zone_id
     LEFT JOIN delivery_payments p ON p.delivery_id = d.id
     WHERE d.customer_id = ?
     ORDER BY d.created_at DESC`,
    [customerId]
  );
  return rows;
}
 
// used by the tracking page: one delivery with names, payment and courier
async function getDeliveryDetail(id) {
  const [rows] = await pool.query(
    `SELECT d.*,
            t.name  AS vehicle,
            pz.name AS pickup_zone, pz.restricted_note AS pickup_note,
            dz.name AS dropoff_zone, dz.restricted_note AS dropoff_note,
            p.method AS payment_method, p.status AS payment_status,
            c.name  AS courier_name, c.phone AS courier_phone
     FROM deliveries d
     JOIN delivery_types t ON t.id = d.delivery_type_id
     JOIN zones pz ON pz.id = d.pickup_zone_id
     JOIN zones dz ON dz.id = d.dropoff_zone_id
     LEFT JOIN delivery_payments p ON p.delivery_id = d.id
     LEFT JOIN users c ON c.id = d.courier_id
     WHERE d.id = ?`,
    [id]
  );
  return rows[0];
}
 
// every job no courier holds right now (waiting for pickup, or waiting at the warehouse)
async function getOpenJobs() {
  const [rows] = await pool.query(
    `SELECT d.id, d.tracking_no, d.status, d.delivery_type_id,
            d.pickup_zone_id, d.dropoff_zone_id, d.scheduled_pickup_at,
            t.name AS vehicle,
            pz.name AS pickup_zone, dz.name AS dropoff_zone,
            p.method AS payment_method, p.status AS payment_status
     FROM deliveries d
     JOIN delivery_types t ON t.id = d.delivery_type_id
     JOIN zones pz ON pz.id = d.pickup_zone_id
     JOIN zones dz ON dz.id = d.dropoff_zone_id
     LEFT JOIN delivery_payments p ON p.delivery_id = d.id
     WHERE d.status IN ('pending', 'at_warehouse')
     ORDER BY d.scheduled_pickup_at`
  );
  return rows;
}
 
// give a job to a courier, only if it's still in the status he saw
async function assignCourier(deliveryId, courierId, fromStatus) {
  const [result] = await pool.query(
    `UPDATE deliveries
     SET courier_id = ?, status = 'assigned'
     WHERE id = ? AND status = ?`,
    [courierId, deliveryId, fromStatus]
  );
  return result.affectedRows; // 1 = he got it, 0 = someone else was faster
}
 
// move a job to a new status, only if it's still in the status we checked
async function updateStatus(deliveryId, fromStatus, toStatus, courierId) {
  const [result] = await pool.query(
    `UPDATE deliveries
     SET status = ?, courier_id = ?
     WHERE id = ? AND status = ?`,
    [toStatus, courierId, deliveryId, fromStatus]
  );
  return result.affectedRows;
}
// parcels waiting at the warehouse for a courier, oldest first
async function getWarehouseParcels() {
  const [rows] = await pool.query(
    `SELECT d.id, d.tracking_no, d.dropoff_zone_id, d.updated_at,
            t.name AS vehicle, dz.name AS dropoff_zone
     FROM deliveries d
     JOIN delivery_types t ON t.id = d.delivery_type_id
     JOIN zones dz ON dz.id = d.dropoff_zone_id
     WHERE d.status = 'at_warehouse'
     ORDER BY d.updated_at`
  );
  return rows;
}
 
// admin orders list: every filter is optional, ? keeps it safe from SQL injection
async function getOrders(filters) {
  const where = [];
  const params = [];
 
  if (filters.status) { where.push('d.status = ?'); params.push(filters.status); }
  if (filters.payment) { where.push('p.status = ?'); params.push(filters.payment); }
  if (filters.zoneId) {
    where.push('(d.pickup_zone_id = ? OR d.dropoff_zone_id = ?)');
    params.push(filters.zoneId, filters.zoneId);
  }
  if (filters.from) { where.push('DATE(d.created_at) >= ?'); params.push(filters.from); }
  if (filters.to) { where.push('DATE(d.created_at) <= ?'); params.push(filters.to); }
 
  const whereSql = where.length ? 'WHERE ' + where.join(' AND ') : '';
 
  const [rows] = await pool.query(
    `SELECT d.id, d.tracking_no, d.status, d.total_fee, d.created_at,
            u.name AS customer, pz.name AS pickup_zone, dz.name AS dropoff_zone,
            p.method AS pay_method, p.status AS pay_status
     FROM deliveries d
     JOIN users u ON u.id = d.customer_id
     JOIN zones pz ON pz.id = d.pickup_zone_id
     JOIN zones dz ON dz.id = d.dropoff_zone_id
     LEFT JOIN delivery_payments p ON p.delivery_id = d.id
     ${whereSql}
     ORDER BY d.created_at DESC`,
    params
  );
  return rows;
}
// courier's own unfinished jobs, with addresses he needs on the road
async function getJobsInHand(courierId) {
  const [rows] = await pool.query(
    `SELECT d.id, d.tracking_no, d.status, d.delivery_type_id,
            d.pickup_zone_id, d.dropoff_zone_id, d.allow_doorstep,
            d.pickup_address, d.dropoff_address,
            d.dropoff_contact_name, d.dropoff_contact_phone,
            pz.name AS pickup_zone, dz.name AS dropoff_zone,
            p.method AS payment_method, p.status AS payment_status, d.total_fee
     FROM deliveries d
     JOIN zones pz ON pz.id = d.pickup_zone_id
     JOIN zones dz ON dz.id = d.dropoff_zone_id
     LEFT JOIN delivery_payments p ON p.delivery_id = d.id
     WHERE d.courier_id = ?
       AND d.status IN ('assigned', 'picked_up', 'out_for_delivery', 'failed')
     ORDER BY d.updated_at`,
    [courierId]
  );
  return rows;
}
